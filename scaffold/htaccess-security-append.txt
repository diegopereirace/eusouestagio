
# BEGIN eusouestagio-security
# Bloqueia documentação e scripts expostos no docroot (README.md, LICENSE.txt, *.sh, etc.).
# Mantido via composer scaffold append — não remover os marcadores BEGIN/END.
<FilesMatch "(?i)^(.*\.(md|markdown|sh)|((README|CHANGELOG|COPYRIGHT|INSTALL|LICENSE|MAINTAINERS|PATCHES|UPDATE|UPGRADE)(\..*)?))$">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
  </IfModule>
</FilesMatch>

# Nega Composer vendor/ (ex.: vendor/bin/drush); permite core/assets/vendor (JS público do Drupal).
# Reforça bloqueio de *.sh.
<IfModule mod_rewrite.c>
  RewriteEngine on
  RewriteRule (?i)^(?!core/assets/)(?:.+/)?vendor(?:/|$) - [F]
  RewriteRule (?i)\.sh$ - [F]
</IfModule>
# END eusouestagio-security
